if(BUILD_SIMULATOR)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.30.11 GIT_SHALLOW TRUE)
 set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
 set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
 set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
 set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
 FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.2.2 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(SDL2 lvgl)
 # LVGL 9.2.2 adds DLL data import/export attributes on MSVC even for a
 # static library. Static consumers must reference the actual font/class data.
 get_target_property(ARCANA_LVGL_TYPE lvgl TYPE)
 if(MSVC AND ARCANA_LVGL_TYPE STREQUAL "STATIC_LIBRARY")
  foreach(ARCANA_LVGL_PROPERTY COMPILE_DEFINITIONS INTERFACE_COMPILE_DEFINITIONS)
   get_target_property(ARCANA_LVGL_DEFINITIONS lvgl ${ARCANA_LVGL_PROPERTY})
   list(FILTER ARCANA_LVGL_DEFINITIONS EXCLUDE REGEX "^LV_ATTRIBUTE_EXTERN_DATA=")
   set_property(TARGET lvgl PROPERTY ${ARCANA_LVGL_PROPERTY} "${ARCANA_LVGL_DEFINITIONS}")
  endforeach()
 endif()
