cmake_minimum_required(VERSION 3.24)
project(MtgLifeCounter VERSION 0.11.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
option(BUILD_SIMULATOR "Build LVGL/SDL desktop application" ON)
add_library(game_core src/core/GameState.cpp src/core/Serialization.cpp src/core/SessionPersistence.cpp src/core/RandomTools.cpp)
target_include_directories(game_core PUBLIC src)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE game_core)
target_include_directories(core_tests PRIVATE simulator)
if(MINGW)
 target_link_options(core_tests PRIVATE -static)
endif()
add_test(NAME game_engine COMMAND core_tests)
add_executable(storage_tests tests/storage_tests.cpp simulator/FileStorage.cpp)
target_include_directories(storage_tests PRIVATE simulator)
target_link_libraries(storage_tests PRIVATE game_core)
if(MINGW)
 target_link_options(storage_tests PRIVATE -static)
endif()
add_test(NAME saved_games COMMAND storage_tests)
if(BUILD_SIMULATOR)
 set(SDL_SHARED OFF CACHE BOOL "" FORCE)
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 set(SDL_TEST OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.30.11 GIT_SHALLOW TRUE)
 set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
 set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
 set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
 set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
 FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.2.2 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(SDL2 lvgl)
 add_executable(mtg_simulator simulator/main.cpp simulator/FileStorage.cpp simulator/SimulatorHardware.cpp src/ui/AppUI.cpp src/ui/AppUIDial.cpp src/ui/AppUITools.cpp src/ui/Widgets/ToolIcons.cpp src/assets/life_digits.c src/assets/mtg_symbols.c src/assets/heart_symbol.c src/assets/tab_font.c)
 target_include_directories(mtg_simulator PRIVATE src simulator)
 target_compile_definitions(mtg_simulator PRIVATE SDL_MAIN_HANDLED)
 target_link_libraries(mtg_simulator PRIVATE game_core lvgl SDL2::SDL2-static)
 if(MINGW)
  target_link_options(mtg_simulator PRIVATE -static)
 endif()
 add_executable(ui_tests tests/ui_tests.cpp src/ui/AppUI.cpp src/ui/AppUIDial.cpp src/ui/AppUITools.cpp src/ui/Widgets/ToolIcons.cpp src/assets/life_digits.c src/assets/mtg_symbols.c src/assets/heart_symbol.c src/assets/tab_font.c)
 target_include_directories(ui_tests PRIVATE src simulator)
 target_link_libraries(ui_tests PRIVATE game_core lvgl)
 if(MINGW)
  target_link_options(ui_tests PRIVATE -static)
 endif()
 add_test(NAME counter_ui COMMAND ui_tests)
 add_test(NAME simulator_smoke COMMAND mtg_simulator --smoke-test)
 add_test(NAME process_recovery COMMAND ${CMAKE_COMMAND} -DSIMULATOR=$<TARGET_FILE:mtg_simulator> -DSAVE_FILE=${CMAKE_CURRENT_BINARY_DIR}/recovery-session.json -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/process_recovery.cmake)
 set_tests_properties(game_engine saved_games counter_ui simulator_smoke process_recovery PROPERTIES TIMEOUT 30)
 set_tests_properties(simulator_smoke process_recovery PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
endif()
